-- Para a tabela Vendas importada de artifacts/vendas_tratadas.csv.
Receita = SUM(Vendas[valor_total])
Transacoes = DISTINCTCOUNT(Vendas[id_venda])
Unidades = SUM(Vendas[quantidade])
Clientes = DISTINCTCOUNT(Vendas[id_cliente])
Ticket por Transacao = DIVIDE([Receita], [Transacoes])
Receita por Unidade = DIVIDE([Receita], [Unidades])
Custo Estimado = SUM(Vendas[custo_estimado])
Margem Bruta Estimada = [Receita] - [Custo Estimado]
Margem Bruta Estimada Percentual = DIVIDE([Margem Bruta Estimada], [Receita])
-- YoY: requer Dim_Calendario marcada como tabela de datas e relacionamento com Vendas[data].
Receita Ano Anterior = CALCULATE([Receita], SAMEPERIODLASTYEAR(Dim_Calendario[data]))
Receita YoY Percentual = DIVIDE([Receita] - [Receita Ano Anterior], [Receita Ano Anterior])
